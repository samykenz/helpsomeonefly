/* ===================== CONFIG — edit this ===================== */
const CONFIG = {
  ticker: "$TICKER",
  buyUrl: "#",                     // pump.fun / DEX link
  contract: "",                    // token CA
  feeWallet: "",                   // public Solana wallet that receives creator fees
  consent: "",                     // e.g. "Shared with the blessing of the Robinson and Sharp families."
  // Every donation sent. sol = amount, tx = Solana signature, receipt = Wings of Hope receipt link.
  donations: [
    // { date: "2026-11-10", sol: 12.5, tx: "5abc...", receipt: "https://..." },
  ],
  rpc: "https://solana-rpc.publicnode.com",   // swap for a Helius URL at launch
  refreshSec: 30
};
/* ============================================================== */

const $ = id => document.getElementById(id);
const usd = n => "$" + n.toLocaleString("en-US",{maximumFractionDigits: n < 1000 ? 2 : 0, minimumFractionDigits: n < 1000 ? 2 : 0});
const sol = n => n.toLocaleString(undefined,{maximumFractionDigits:3}) + " SOL";

document.querySelectorAll("[data-cfg]").forEach(el => { const v = CONFIG[el.dataset.cfg]; if (v) el.textContent = v; });
$("buyBtn").href = CONFIG.buyUrl;
if (CONFIG.contract) $("ca").textContent = CONFIG.contract;
if (CONFIG.consent) $("consentLine").textContent = CONFIG.consent;
if (CONFIG.feeWallet) $("walletLink").href = "https://solscan.io/account/" + CONFIG.feeWallet;
else $("walletLink").style.display = "none";
$("copyBtn").onclick = () => {
  if (!CONFIG.contract) return;
  navigator.clipboard?.writeText(CONFIG.contract);
  $("copyBtn").textContent = "Copied"; setTimeout(() => $("copyBtn").textContent = "Copy", 1400);
};

const donatedSol = CONFIG.donations.reduce((s,t) => s + t.sol, 0);
if (CONFIG.donations.length) {
  $("ledgerBody").innerHTML = CONFIG.donations.slice().reverse().map(t =>
    `<tr><td>${t.date}</td><td>${sol(t.sol)}</td><td><a href="https://solscan.io/tx/${t.tx}" target="_blank" rel="noopener">${t.tx.slice(0,6)}…${t.tx.slice(-4)} ↗</a></td><td>${t.receipt ? `<a href="${t.receipt}" target="_blank" rel="noopener">View ↗</a>` : "—"}</td></tr>`).join("");
}

let shown = 0;
function animateTo(target){
  const start = shown, t0 = performance.now(), dur = 1600;
  const step = now => {
    const k = Math.min(1,(now - t0)/dur), e = 1 - Math.pow(1-k,4);
    shown = start + (target - start) * e;
    $("raised").textContent = usd(shown);
    if (k < 1) requestAnimationFrame(step);
  };
  requestAnimationFrame(step);
}

async function rpcBalance(addr){
  const r = await fetch(CONFIG.rpc,{method:"POST",headers:{"Content-Type":"application/json"},
    body:JSON.stringify({jsonrpc:"2.0",id:1,method:"getBalance",params:[addr,{commitment:"confirmed"}]})});
  const j = await r.json(); if (j.error) throw new Error(j.error.message);
  return j.result.value / 1e9;
}
async function solPrice(){
  const r = await fetch("https://api.coingecko.com/api/v3/simple/price?ids=solana&vs_currencies=usd");
  return (await r.json()).solana.usd;
}

function setStatus(live, text, note=""){
  $("dot").classList.toggle("off", !live);
  $("status").textContent = text;
  $("notice").textContent = note;
}

// new fees arriving: float "+X SOL" and light up the bar
function celebrate(deltaSol){
  const b = $("bump"); b.textContent = "+" + sol(deltaSol);
  b.classList.remove("go"); void b.offsetWidth; b.classList.add("go");
  $("bar").classList.add("glow"); setTimeout(() => $("bar").classList.remove("glow"), 1400);
}

let lastPrice = 0, lastBal = null;
async function refresh(){
  if (!CONFIG.feeWallet) {
    setStatus(false, "Goes live at launch", "The tracker starts the moment the fee wallet is added. No fake numbers before then.");
    return;
  }
  try {
    const [bal, price] = await Promise.all([rpcBalance(CONFIG.feeWallet), solPrice().catch(() => lastPrice)]);
    if (price) lastPrice = price;
    if (lastBal !== null && bal > lastBal + 1e-6) celebrate(bal - lastBal);
    lastBal = bal;
    const totalSol = bal + donatedSol;
    // the bar is the whole fund: blue = already donated, orange = waiting for the next donation
    const sentPct = totalSol ? donatedSol / totalSol * 100 : 0;
    $("segSent").style.width = sentPct + "%";
    $("segPend").style.width = (totalSol ? 100 - sentPct : 0) + "%";
    $("totalSol").textContent = sol(totalSol);
    $("lSent").textContent = usd(donatedSol * lastPrice);
    $("lPend").textContent = usd(bal * lastPrice);
    $("updated").textContent = new Date().toLocaleTimeString([], {hour:"2-digit",minute:"2-digit",second:"2-digit"});
    animateTo(totalSol * lastPrice);
    setStatus(true, "Live from Solana");
  } catch (e) {
    setStatus(false, "Reconnecting…", "Couldn't reach Solana. Retrying shortly.");
  }
}
refresh();
setInterval(refresh, CONFIG.refreshSec * 1000);
